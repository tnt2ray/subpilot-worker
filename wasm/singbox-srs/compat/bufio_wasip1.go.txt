package bufio

import (
	"errors"
	"io"

	"github.com/sagernet/sing/common/buf"
	M "github.com/sagernet/sing/common/metadata"
	N "github.com/sagernet/sing/common/network"
)

// The SRS compiler does not use sing's socket-specific fast paths. WASI does
// not provide the Unix syscalls those helpers require, so keep the generic
// package API available and leave those paths disabled.
type syscallVectorisedWriterFields struct{}

func (w *SyscallVectorisedWriter) WriteVectorised(buffers []*buf.Buffer) error {
	writer, ok := w.upstream.(io.Writer)
	if !ok {
		buf.ReleaseMulti(buffers)
		return errors.New("vectorised writes are unavailable on WASI")
	}
	return (&BufferedVectorisedWriter{upstream: writer}).WriteVectorised(buffers)
}

func (w *SyscallVectorisedPacketWriter) WriteVectorisedPacket(buffers []*buf.Buffer, _ M.Socksaddr) error {
	buf.ReleaseMulti(buffers)
	return errors.New("vectorised packet writes are unavailable on WASI")
}

func createSyscallReadWaiter(io.Reader) (N.ReadWaiter, bool) {
	return nil, false
}

func createVectorisedSyscallReadWaiter(io.Reader) (N.VectorisedReadWaiter, bool) {
	return nil, false
}

func createSyscallPacketReadWaiter(N.PacketReader) (N.PacketReadWaiter, bool) {
	return nil, false
}
